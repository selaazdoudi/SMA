################################################################################
##### PARAMETRES ###############################################################

H_MIN_SEQ   <- -6    
H_MAX       <- 72    
H_DEB_CLUST <- 1     

N_PAIRES    <- 5000  
GRAINE      <- 2019

VAR_ANS <- "y_ns_max"
UTILISER_RETRAITE <- FALSE
SEUIL_SECRET <- 11

################################################################################
##### 1 & 2. CHARGEMENT ET TIRAGE PAR PAIRE ####################################
################################################################################

apparie <- read_parquet(file.path(dir_out, "echantillon_apparie_2019.parquet")) %>%
  select(id_spell, id_midas, paire, traite, groupe, deb_mois,
         SEXE, qualif_classe, niv_dip, duree_cat,
         age_fin_ctt_redresse, log_sjr)

verif_paires <- apparie %>%
  group_by(paire) %>%
  summarise(n = n(), n_traites = sum(traite), .groups = "drop")

stopifnot(all(verif_paires$n == 2 & verif_paires$n_traites == 1))

set.seed(GRAINE)
paires_tirees <- verif_paires %>%
  slice_sample(n = min(N_PAIRES, nrow(verif_paires))) %>%
  pull(paire)

ech <- apparie %>% filter(paire %in% paires_tirees)

cat("Paires tirees :", length(paires_tirees),
    "| sequences :", nrow(ech),
    "| dont traites :", sum(ech$traite), "\n")

################################################################################
##### 3. CONSTRUCTION DES ETATS ################################################
################################################################################

panel_lazy <- open_dataset(file.path(dir_out, "panel_pour_appariement_2019.parquet"))

panel_ech <- panel_lazy %>%
  select(id_spell, h, y_salarie, y_ns_min, y_ns_max,
         f_indemnise, f_minima, f_retraite) %>%
  filter(h >= H_MIN_SEQ, h <= H_MAX) %>%
  collect() %>%
  inner_join(ech %>% select(id_spell, paire, traite), by = "id_spell")

construire_etats <- function(df, var_ans) {
  df %>%
    mutate(
      y_ans = .data[[var_ans]],
      etat_seq = case_when(
        y_ans == 1 & y_salarie == 1                          ~ "ANS_SAL",
        y_ans == 1 & f_indemnise == 1                        ~ "ANS_INDEM",
        y_ans == 1                                           ~ "ANS_SEULE",
        y_salarie == 1 & (f_indemnise == 1 | f_minima == 1)  ~ "EMP_AIDES",
        y_salarie == 1                                       ~ "EMP_SAL",
        f_indemnise == 1                                     ~ "CHOM_IND",
        f_minima == 1                                        ~ "AUTRE_MINIMA",
        UTILISER_RETRAITE & f_retraite == 1                  ~ "AUTRE_RETRAITE",
        TRUE                                                 ~ "NON_OBSERVE"
      )
    )
}

panel_etats <- construire_etats(panel_ech, VAR_ANS)

################################################################################
##### 4. OBJETS SEQUENCE #######################################################
################################################################################

alphabet_etats <- c("EMP_SAL", "EMP_AIDES", "ANS_SAL", "ANS_INDEM", "ANS_SEULE", "CHOM_IND", "AUTRE_MINIMA", "NON_OBSERVE")
labels_etats   <- c("Emploi salarie", "Emploi salarie avec aides", "Non salarie + salarie", "Non salarie indemnise", "Non salarie seul", "Chomage indemnise", "Minima sociaux", "Non observe")
palette_couleurs <- c("#1A9850", "#A6D96A", "#8C510A", "#D73027", "#F46D43", "#4393C3", "#984EA3", "#E0E0E0")

if (UTILISER_RETRAITE) {
  alphabet_etats   <- append(alphabet_etats, "AUTRE_RETRAITE", after = 7)
  labels_etats     <- append(labels_etats, "Retraite", after = 7)
  palette_couleurs <- append(palette_couleurs, "#4D4D4D", after = 7)
}

etats_presents   <- intersect(alphabet_etats, unique(panel_etats$etat_seq))
idx              <- match(etats_presents, alphabet_etats)
labels_presents  <- labels_etats[idx]
couleurs_presentes <- palette_couleurs[idx]

faire_seq <- function(df, h_deb, h_fin) {
  wide <- df %>%
    filter(h >= h_deb, h <= h_fin) %>%
    select(id_spell, h, etat_seq) %>%
    arrange(id_spell, h) %>%
    pivot_wider(names_from = h, values_from = etat_seq, names_prefix = "m", names_sort = TRUE)

  cols <- paste0("m", h_deb:h_fin)
  obj  <- seqdef(wide, var = cols, alphabet = etats_presents, labels = labels_presents, cpal = couleurs_presentes, id = wide$id_spell, xtstep = 6)
  list(seq = obj, ids = wide$id_spell)
}

# (a) sequences pour la TYPOLOGIE (Post-rupture)
s_post <- faire_seq(panel_etats, H_DEB_CLUST, H_MAX)
seq_post <- s_post$seq

# (b) sequences pour le CHRONOGRAMME descriptif (Pre+Post)
s_tot <- faire_seq(panel_etats, H_MIN_SEQ, H_MAX)
seq_tot <- s_tot$seq

info_post <- tibble(id_spell = s_post$ids) %>%
  left_join(ech, by = "id_spell")

# SEPARATION DES OBJETS SEQUENCES
seq_post_rci <- seq_post[info_post$traite == 1, ]
info_post_rci <- info_post %>% filter(traite == 1)

seq_post_tem <- seq_post[info_post$traite == 0, ]
info_post_tem <- info_post %>% filter(traite == 0)

################################################################################
##### 5. CHRONOGRAMMES GLOBAUX COMPARES ########################################
################################################################################

grp_lbl_tot  <- factor(ech$traite[match(s_tot$ids, ech$id_spell)], levels = c(1, 0), labels = c("Rupture conventionnelle", "Fin de CDI hors RC"))

png(file.path(dir_res, "fig1_chronogramme_compare_global.png"), width = 2400, height = 1100, res = 180)
seqdplot(seq_tot, group = grp_lbl_tot, border = NA, with.legend = "right", cex.legend = 0.75, main = NULL, xlab = "Mois depuis la fin de contrat (h = 0 : mois de la rupture)")
dev.off()

################################################################################
##### 6. DISTANCES ET TYPOLOGIES SEPAREES ######################################
################################################################################

METHODE_COUTS <- "TRATE"

# --- A. TYPOLOGIE RCI ---
couts_rci <- seqsubm(seq_post_rci, method = METHODE_COUTS, cval = if (METHODE_COUTS == "CONSTANT") 2 else NULL)
diss_rci  <- seqdist(seq_post_rci, method = "OM", indel = 1, sm = couts_rci)
arbre_rci <- hclust(as.dist(diss_rci), method = "ward.D2")

qualite_rci <- as.clustrange(arbre_rci, diss = diss_rci, ncluster = 10)
K_rci <- which.max(qualite_rci$stats[, "ASW"]) + 1
cat("\n[RCI] k retenu (ASW maximal) :", K_rci, "\n")

classe_rci <- cutree(arbre_rci, k = K_rci)
info_post_rci$classe <- factor(classe_rci, levels = 1:K_rci, labels = paste0("RCI_Cl ", 1:K_rci))

# --- B. TYPOLOGIE TEMOINS ---
couts_tem <- seqsubm(seq_post_tem, method = METHODE_COUTS, cval = if (METHODE_COUTS == "CONSTANT") 2 else NULL)
diss_tem  <- seqdist(seq_post_tem, method = "OM", indel = 1, sm = couts_tem)
arbre_tem <- hclust(as.dist(diss_tem), method = "ward.D2")

qualite_tem <- as.clustrange(arbre_tem, diss = diss_tem, ncluster = 10)
K_tem <- which.max(qualite_tem$stats[, "ASW"]) + 1
cat("\n[TEMOINS] k retenu (ASW maximal) :", K_tem, "\n")

classe_tem <- cutree(arbre_tem, k = K_tem)
info_post_tem$classe <- factor(classe_tem, levels = 1:K_tem, labels = paste0("TEM_Cl ", 1:K_tem))

# Fusion des informations
info_post_final <- bind_rows(info_post_rci, info_post_tem)
rm(diss_rci, diss_tem); gc()

  group_by(groupe) %>%
