construire_etats <- function(df, var_ans) {
  df %>%
    mutate(
      y_ans = .data[[var_ans]],
      etat_seq = case_when(
        y_ans == 1 & y_salarie == 1                          ~ "ANS_SAL",
        y_ans == 1 & f_indemnise == 1                        ~ "ANS_INDEM",
        y_ans == 1                                           ~ "ANS_SEULE",
        y_salarie == 1 & (f_indemnise == 1 | f_minima == 1)  ~ "EMP_AIDES",
        y_salarie == 1                                       ~ "EMP_SAL",
        f_indemnise == 1                                     ~ "CHOM_IND",
        f_minima == 1                                        ~ "AUTRE_MINIMA",
        UTILISER_RETRAITE & f_retraite == 1                  ~ "AUTRE_RETRAITE",
        TRUE                                                 ~ "NON_OBSERVE"
      )
    )
}

faire_seq <- function(df, h_deb, h_fin) {
  wide <- df %>%
    filter(h >= h_deb, h <= h_fin) %>%
    select(id_spell, h, etat_seq) %>%
    arrange(id_spell, h) %>%
    pivot_wider(names_from = h, values_from = etat_seq,
                names_prefix = "m", names_sort = TRUE)

  stopifnot(!any(is.na(wide)))

  cols <- paste0("m", h_deb:h_fin)
  obj  <- seqdef(wide, var = cols,
                 alphabet = etats_presents,
                 labels   = labels_presents,
                 cpal     = couleurs_presentes,
                 id       = wide$id_spell,
                 xtstep   = 6)
  list(seq = obj, ids = wide$id_spell)
}

##### 8. SORTIE PRINCIPALE : CLASSE x TRAITEMENT ###############################
################################################################################
# Les paires etant 1:1, on teste la difference de composition PAR PAIRE, ce qui
# tient compte de l'appariement (equivalent apparie d'une difference de
# proportions ; version multi-classes du test de McNemar).

repartition <- info_post %>%
  count(traite, classe) %>%
  group_by(traite) %>%
  mutate(pct = 100 * n / sum(n)) %>%
  ungroup() %>%
  mutate(groupe = if_else(traite == 1, "RCI", "Temoins"))

tableau_croise <- repartition %>%
  select(groupe, classe, n, pct) %>%
  pivot_wider(names_from = groupe, values_from = c(n, pct))

# Difference appariee, classe par classe
diff_appariee <- map_dfr(levels(info_post$classe), function(cl) {
  par_paire <- info_post %>%
    mutate(ind = as.integer(classe == cl)) %>%
    select(paire, traite, ind) %>%
    pivot_wider(names_from = traite, values_from = ind,
                names_prefix = "t") %>%
    filter(!is.na(t1), !is.na(t0)) %>%
    mutate(d = t1 - t0)

  n_p <- nrow(par_paire)
  m   <- mean(par_paire$d)
  se  <- sd(par_paire$d) / sqrt(n_p)

  tibble(
    classe      = cl,
    n_paires    = n_p,
    pct_RCI     = 100 * mean(par_paire$t1),
    pct_temoins = 100 * mean(par_paire$t0),
    diff_pp     = 100 * m,
    ic_bas_pp   = 100 * (m - 1.96 * se),
    ic_haut_pp  = 100 * (m + 1.96 * se),
    p_value     = 2 * pnorm(-abs(m / se))
  )
})

cat("\n=== SORTIE PRINCIPALE : composition des trajectoires ===\n")
