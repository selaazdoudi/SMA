faire_seq <- function(df, h_deb, h_fin) {
  wide <- df %>%
    filter(h >= h_deb, h <= h_fin) %>%
    select(id_spell, h, etat_seq) %>%
    arrange(id_spell, h) %>%
    pivot_wider(names_from = h, values_from = etat_seq, names_prefix = "m", names_sort = TRUE)

  cols <- paste0("m", h_deb:h_fin)
  seqdef(wide, var = cols, alphabet = etats_presents, labels = labels_presents, cpal = couleurs_presentes, id = wide$id_spell, xtstep = 6)
}

##### 4. TYPOLOGIES SEPAREES (OPTIMAL MATCHING + CAH + CONSOLIDATION PAM) ######
################################################################################

faire_typologie <- function(sequences, k_choisi, prefixe) {
  cat("\nCalcul de la typologie pour :", prefixe, "\n")
  
  # Matrice de distances (OM avec coûts constants de type seqhandbook)
  couts <- seqsubm(sequences, method = "CONSTANT", cval = 2)
  diss  <- seqdist(sequences, method = "OM", indel = 1.5, sm = couts)
  
  # CAH de Ward
  arbre <- hclust(as.dist(diss), method = "ward.D2")
  part_cah <- cutree(arbre, k = k_choisi)
  
  # Consolidation par l'algorithme PAM (K-Medoids) pour optimiser le R2
  part_pam <- wcKMedoids(diss, k = k_choisi, initialclust = part_cah, cluster.only = TRUE)
  
  # Formatage en facteur propre
  facteur_classe <- factor(as.numeric(as.factor(part_pam)), levels = 1:k_choisi, labels = paste0(prefixe, "_Cl", 1:k_choisi))
  
  list(classe = facteur_classe, diss = diss)
}

decrire_classes <- function(seq_obj, vecteur_classes, diss_matrix, prefixe) {
  cat("\nGénération des graphiques de description pour :", prefixe, "\n")
  
  # 1. Chronogrammes par classe
  exporter_graphe(paste0("desc_", prefixe, "_1_chronogrammes.png"), 
                  seqdplot(seq_obj, group = vecteur_classes, border = NA, with.legend = "right", main = paste(prefixe, "- Répartition des états")))
  
  # 2. Temps moyen passé dans chaque état
  exporter_graphe(paste0("desc_", prefixe, "_2_temps_moyen.png"), 
                  seqmtplot(seq_obj, group = vecteur_classes, with.legend = "right", main = paste(prefixe, "- Temps moyen par état")))
  
  # 3. Séquences représentatives
  exporter_graphe(paste0("desc_", prefixe, "_3_seq_representatives.png"), 
                  seqrplot(seq_obj, group = vecteur_classes, dist.matrix = diss_matrix, border = NA, coverage = 0.3, with.legend = "right", main = paste(prefixe, "- Séquences représentatives")))
  
  # 4. Entropie transversale (Incertitude au fil du temps)
  exporter_graphe(paste0("desc_", prefixe, "_4_entropie.png"), 
                  seqHtplot(seq_obj, group = vecteur_classes, main = paste(prefixe, "- Entropie transversale")))
  
  # 5. Calcul des indicateurs de complexité (Complexité et nombre de transitions)
  complexite  <- seqici(seq_obj)
  nb_episodes <- seqtransn(seq_obj) + 1
  
  stats_complexite <- tibble(
    classe = vecteur_classes,
    complexite = as.numeric(complexite),
    nb_episodes = as.numeric(nb_episodes)
  ) %>%
    group_by(classe) %>%
    summarise(
      complexite_moyenne = mean(complexite, na.rm = TRUE),
      nb_episodes_moyen = mean(nb_episodes, na.rm = TRUE),
      .groups = "drop"
    )
  
  write_csv(stats_complexite, file.path(dir_res, paste0("desc_", prefixe, "_5_complexite.csv")))
}
